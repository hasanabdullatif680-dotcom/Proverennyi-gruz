<script>
(function(){
  const tg = window.Telegram?.WebApp;
  if (tg) { tg.ready(); tg.expand(); }
  const tgUser = tg?.initDataUnsafe?.user;
  const userName = tgUser?.first_name || 'Гость';

  const STORAGE = 'freightglass_v2';
  let state = {
    items: [],
    favs: [],
    role: 'carrier',
    view: 'list',
    sort: 'new',
    notif: true,
    addedCount: 0
  };

  // ---------- загрузка ----------
  function load(){
    try {
      const raw = localStorage.getItem(STORAGE);
      if (raw) state = Object.assign(state, JSON.parse(raw));
    } catch(e){}
    if (!state.items.length && !localStorage.getItem(STORAGE)) seed();
    updateBadge();
  }

  function save(){ localStorage.setItem(STORAGE, JSON.stringify(state)); }

  function seed(){
    const now = Date.now();
    state.items = [
      {id:now-1000,from:'Москва',to:'Санкт-Петербург',type:'Тент',weight:18,price:45000,note:'',owner:'demo',role:'shipper',ts:now-1000},
      {id:now-2000,from:'Казань',to:'Екатеринбург',type:'Рефрижератор',weight:12,price:62000,note:'Температурный режим -18',owner:'demo',role:'shipper',ts:now-2000},
      {id:now-3000,from:'Новосибирск',to:'Красноярск',type:'Фура',weight:22,price:38000,note:'',owner:'demo',role:'shipper',ts:now-3000},
      {id:now-4000,from:'Москва',to:'Нижний Новгород',type:'Борт',weight:5,price:18000,note:'',owner:'demo',role:'shipper',ts:now-4000},
      {id:now-5000,from:'Екатеринбург',to:'Москва',type:'Фура',weight:20,price:55000,note:'Обратный рейс',owner:'demo',role:'carrier',ts:now-5000}
    ];
    save();
  }

  function updateBadge(){
    document.getElementById('userBadge').textContent = userName;
    const initial = userName.charAt(0).toUpperCase();
    document.getElementById('avatar').textContent = initial;
    document.getElementById('profileName').textContent = userName;
  }

  // ---------- утилиты ----------
  function esc(s){ return String(s||'').replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c])); }
  function fmt(n){ return Number(n).toLocaleString('ru-RU'); }

  function showToast(msg){
    const t = document.getElementById('toast');
    t.textContent = msg; t.classList.add('show');
    clearTimeout(t._tm);
    t._tm = setTimeout(()=>t.classList.remove('show'), 2000);
  }

  // ---------- фильтрация/сортировка ----------
  function baseList(){
    let list = state.items.slice();

    if (state.view === 'list'){
      if (state.role === 'carrier') list = list.filter(i=>i.role==='shipper');
      else list = list.filter(i=>i.role==='carrier');
    } else if (state.view === 'my'){
      list = list.filter(i=>i.owner==='me');
    } else if (state.view === 'fav'){
      list = list.filter(i=>state.favs.includes(i.id));
    }
    return list;
  }

  function applyFilters(list){
    const search = (document.getElementById('fSearch')?.value||'').toLowerCase();
    const from = (document.getElementById('fFrom')?.value||'').toLowerCase();
    const to = (document.getElementById('fTo')?.value||'').toLowerCase();
    const type = document.getElementById('fType')?.value||'';
    const weight = document.getElementById('fWeight')?.value||'';

    return list.filter(i=>{
      if (search && !(i.from+i.to+i.type+(i.note||'')).toLowerCase().includes(search)) return false;
      if (from && !i.from.toLowerCase().includes(from)) return false;
      if (to && !i.to.toLowerCase().includes(to)) return false;
      if (type && i.type !== type) return false;
      if (weight){
        const w = +i.weight||0;
        if (weight==='light' && w>5) return false;
        if (weight==='medium' && (w<=5||w>20)) return false;
        if (weight==='heavy' && w<=20) return false;
      }
      return true;
    });
  }

  function sortList(list){
    const s = state.sort;
    const arr = list.slice();
    if (s==='priceAsc') arr.sort((a,b)=>a.price-b.price);
    else if (s==='priceDesc') arr.sort((a,b)=>b.price-a.price);
    else if (s==='weightAsc') arr.sort((a,b)=>a.weight-b.weight);
    else if (s==='weightDesc') arr.sort((a,b)=>b.weight-a.weight);
    else arr.sort((a,b)=>b.ts-a.ts);
    return arr;
  }

  // ---------- рендер карточек ----------
  function cardHTML(it, mode){
    const isFav = state.favs.includes(it.id);
    const isMine = it.owner==='me';
    return `
    <div class="card glass" data-id="${it.id}">
      <div class="top">
        <div class="route">
          <span class="city">${esc(it.from)}</span>
          <span class="arrow">→</span>
          <span class="city">${esc(it.to)}</span>
        </div>
        <button class="fav ${isFav?'on':''}" data-act="fav" data-id="${it.id}">${isFav?'★':'☆'}</button>
      </div>
      <div class="meta">
        <span class="tag">${esc(it.type)}</span>
        <span class="tag">${it.weight} т</span>
        <span class="tag">${it.role==='shipper'?'📦 Груз':'🚛 Машина'}</span>
        ${it.note?`<span class="tag">💬 ${esc(it.note)}</span>`:''}
      </div>
      <div class="price">${fmt(it.price)} ₽ <small>${it.role==='shipper'?'за перевозку':'за рейс'}</small></div>
      <div class="actions">
        ${isMine ? `
          <button class="btn-edit" data-act="edit" data-id="${it.id}">✏️ Изменить</button>
          <button class="btn-danger" data-act="del" data-id="${it.id}">Удалить</button>
        ` : `
          <button class="btn-primary" data-act="contact" data-id="${it.id}">Связаться</button>
          <button class="btn-ghost" data-act="fav" data-id="${it.id}">${isFav?'★ В избранном':'☆ В избранное'}</button>
        `}
      </div>
    </div>`;
  }

  function render(){
    // лента
    if (state.view==='list'){
      const list = sortList(applyFilters(baseList()));
      document.getElementById('countLabel').textContent = list.length+' записей';
      document.getElementById('listContainer').innerHTML = list.length
        ? list.map(i=>cardHTML(i,'list')).join('')
        : `<div class="empty glass"><div class="icon">📭</div>Ничего не найдено<br>Попробуй изменить фильтры</div>`;
    }
    // мои
    if (state.view==='my'){
      const list = sortList(baseList());
      document.getElementById('myCount').textContent = 'Мои записи: '+list.length;
      document.getElementById('myContainer').innerHTML = list.length
        ? list.map(i=>cardHTML(i,'my')).join('')
        : `<div class="empty glass"><div class="icon">📁</div>Ты ещё ничего не добавил<br>Нажми «+» внизу</div>`;
    }
    // избранное
    if (state.view==='fav'){
      const list = sortList(baseList());

  // ---------- навигация ----------
  function switchView(v){
    state.view = v; save();
    ['list','my','fav','profile'].forEach(k=>{
      document.getElementById('view-'+k).classList.toggle('hidden', k!==v);
    });
    document.querySelectorAll('.bottom-nav button').forEach(b=>{
      b.classList.toggle('active', b.dataset.view===v);
    });
    render();
  }

  document.querySelectorAll('.bottom-nav button').forEach(b=>{
    b.addEventListener('click', ()=>switchView(b.dataset.view));
  });

  // роли
  document.querySelectorAll('.role-switch button').forEach(b=>{
    b.addEventListener('click', ()=>{
      document.querySelectorAll('.role-switch button').forEach(x=>x.classList.remove('active'));
      b.classList.add('active');
      state.role = b.dataset.role; save();
      render();
    });
